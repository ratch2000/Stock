# ----------------------------------------------------------------- data
def to_symbol(t: str) -> str:
    t = t.strip().upper()
    return t if t.endswith(".BK") else f"{t}.BK"


@st.cache_data(ttl=3600, show_spinner=False)
def load(symbol: str, period: str = "2y"):
    tk = yf.Ticker(symbol)
    hist = tk.history(period=period, auto_adjust=True)
    try:
        info = tk.info or {}
    except Exception:
        info = {}
    return hist, info


# ----------------------------------------------------------- indicators
def rsi(close, n=14):
    d = close.diff()
    up = d.clip(lower=0).ewm(alpha=1 / n, adjust=False).mean()
    dn = (-d.clip(upper=0)).ewm(alpha=1 / n, adjust=False).mean()
    return 100 - 100 / (1 + up / dn.replace(0, np.nan))


def atr(df, n=14):
    pc = df["Close"].shift()
    tr = pd.concat(
        [df["High"] - df["Low"], (df["High"] - pc).abs(), (df["Low"] - pc).abs()], axis=1
    ).max(axis=1)
    return tr.ewm(alpha=1 / n, adjust=False).mean()


def add_indicators(df):
    df = df.copy()
    df["EMA20"] = df["Close"].ewm(span=20, adjust=False).mean()
    df["SMA50"] = df["Close"].rolling(50).mean()
    df["SMA200"] = df["Close"].rolling(200).mean()
    df["RSI"] = rsi(df["Close"])
    df["ATR"] = atr(df)
    macd = df["Close"].ewm(span=12, adjust=False).mean() - df["Close"].ewm(span=26, adjust=False).mean()
    df["MACD_H"] = macd - macd.ewm(span=9, adjust=False).mean()
    return df


# ---------------------------------------------------------- fundamentals
def g(info, key):
    v = info.get(key)
    return v if isinstance(v, (int, float)) and not pd.isna(v) else None


def score_fundamentals(info, price):
    """คืน list ของ (ตัวชี้วัด, ค่า, คะแนน 0-2, คำอธิบาย) และคะแนนรวม %"""
    is_fin = info.get("sector") == "Financial Services"
    rows = []

    def add(name, val, fmt, pts, why):
        rows.append((name, "-" if val is None else fmt.format(val), pts, why))

    pe = g(info, "trailingPE")
    if pe is None or pe <= 0:
        add("P/E", pe, "{:.1f}", 0, "ไม่มี P/E หรือกำไรติดลบ")
    elif pe < 15:
        add("P/E", pe, "{:.1f}", 2, "ราคาไม่แพงเมื่อเทียบกับกำไร")
    elif pe < 25:
        add("P/E", pe, "{:.1f}", 1, "ราคาอยู่ระดับกลาง")
    else:
        add("P/E", pe, "{:.1f}", 0, "ราคาสูงเมื่อเทียบกับกำไร")

    pb = g(info, "priceToBook")
    if pb is None:
        add("P/B", pb, "{:.2f}", 0, "ไม่มีข้อมูล")
    elif pb < 1.5:
        add("P/B", pb, "{:.2f}", 2, "ซื้อขายใกล้มูลค่าทางบัญชี")
    elif pb < 3:
        add("P/B", pb, "{:.2f}", 1, "มูลค่าทางบัญชีอยู่ระดับกลาง")
    else:
        add("P/B", pb, "{:.2f}", 0, "ราคาสูงกว่ามูลค่าทางบัญชีมาก")

    roe = g(info, "returnOnEquity")
    roe = roe * 100 if roe is not None else None
    if roe is None:
        add("ROE %", roe, "{:.1f}", 0, "ไม่มีข้อมูล")
    elif roe >= 15:
        add("ROE %", roe, "{:.1f}", 2, "ใช้เงินทุนผู้ถือหุ้นสร้างกำไรได้ดี")
    elif roe >= 8:
        add("ROE %", roe, "{:.1f}", 1, "ผลตอบแทนต่อส่วนผู้ถือหุ้นระดับกลาง")
    else:
        add("ROE %", roe, "{:.1f}", 0, "ผลตอบแทนต่อส่วนผู้ถือหุ้นต่ำ")

    if not is_fin:
        de = g(info, "debtToEquity")
        de = de / 100 if de is not None else None
        if de is None:
            add("D/E (เท่า)", de, "{:.2f}", 0, "ไม่มีข้อมูล")
        elif de < 1:
            add("D/E (เท่า)", de, "{:.2f}", 2, "หนี้ต่ำ ฐานะการเงินแข็งแรง")
        elif de < 2:
            add("D/E (เท่า)", de, "{:.2f}", 1, "หนี้อยู่ระดับกลาง")
        else:
            add("D/E (เท่า)", de, "{:.2f}", 0, "หนี้สูง เสี่ยงต่อดอกเบี้ย")

    pm = g(info, "profitMargins")
    pm = pm * 100 if pm is not None else None
    if pm is None:
        add("Net margin %", pm, "{:.1f}", 0, "ไม่มีข้อมูล")
    elif pm >= 12:
        add("Net margin %", pm, "{:.1f}", 2, "อัตรากำไรสุทธิสูง")
    elif pm >= 5:
        add("Net margin %", pm, "{:.1f}", 1, "อัตรากำไรสุทธิปานกลาง")
    else:
        add("Net margin %", pm, "{:.1f}", 0, "อัตรากำไรสุทธิบาง")

    rg = g(info, "revenueGrowth")
    rg = rg * 100 if rg is not None else None
    if rg is None:
        add("รายได้โต % (YoY)", rg, "{:.1f}", 0, "ไม่มีข้อมูล")
    elif rg >= 8:
        add("รายได้โต % (YoY)", rg, "{:.1f}", 2, "รายได้เติบโตดี")
    elif rg >= 0:
        add("รายได้โต % (YoY)", rg, "{:.1f}", 1, "รายได้โตช้า")
    else:
        add("รายได้โต % (YoY)", rg, "{:.1f}", 0, "รายได้หดตัว")

    eg = g(info, "earningsGrowth")
    eg = eg * 100 if eg is not None else None
    if eg is None:
        add("กำไรโต % (YoY)", eg, "{:.1f}", 0, "ไม่มีข้อมูล")
    elif eg >= 10:
        add("กำไรโต % (YoY)", eg, "{:.1f}", 2, "กำไรเติบโตแข็งแรง")
    elif eg >= 0:
        add("กำไรโต % (YoY)", eg, "{:.1f}", 1, "กำไรทรงตัว/โตเล็กน้อย")
    else:
        add("กำไรโต % (YoY)", eg, "{:.1f}", 0, "กำไรลดลง")

    div = g(info, "dividendRate")
    dy = div / price * 100 if div and price else None
    if dy is None:
        add("Dividend yield %", dy, "{:.2f}", 0, "ไม่มีข้อมูลปันผล")
    elif dy >= 4:
        add("Dividend yield %", dy, "{:.2f}", 2, "ปันผลสูง ช่วยหนุนผลตอบแทน")
    elif dy >= 2:
        add("Dividend yield %", dy, "{:.2f}", 1, "ปันผลระดับกลาง")
    else:
        add("Dividend yield %", dy, "{:.2f}", 0, "ปันผลต่ำ")

    pct = sum(r[2] for r in rows) / (2 * len(rows)) * 100
    return rows, pct


# ------------------------------------------------------------- technical
def analyze_technical(df):
    last = df.iloc[-1]
    c, e20, s50, s200 = last["Close"], last["EMA20"], last["SMA50"], last["SMA200"]
    r, a, mh = last["RSI"], last["ATR"], last["MACD_H"]
    pts, notes_pos, notes_neg = 0, [], []

    def chk(cond, p, pos, neg):
        nonlocal pts
        if cond:
            pts += p
            notes_pos.append(pos)
        else:
            notes_neg.append(neg)

    chk(c > s200, 2, "ราคายืนเหนือ SMA200 (แนวโน้มใหญ่เป็นขาขึ้น)", "ราคาต่ำกว่า SMA200 (แนวโน้มใหญ่ยังอ่อน)")
    chk(c > s50, 1, "ราคายืนเหนือ SMA50", "ราคาต่ำกว่า SMA50")
    chk(s50 > s200, 1, "SMA50 อยู่เหนือ SMA200 (Golden Cross)", "SMA50 ต่ำกว่า SMA200 (Death Cross)")
    chk(mh > 0, 1, "MACD histogram เป็นบวก (โมเมนตัมบวก)", "MACD histogram เป็นลบ (โมเมนตัมลบ)")
    if 40 <= r < 70:
        pts += 2
        notes_pos.append(f"RSI {r:.0f} อยู่ในโซนสุขภาพดี")
    elif r < 40:
        pts += 1
        notes_pos.append(f"RSI {r:.0f} เริ่มขายมากเกินไป อาจมีโอกาสรีบาวด์")
    else:
        notes_neg.append(f"RSI {r:.0f} ซื้อมากเกินไป เสี่ยงพักตัว")

    support = df["Low"].tail(60).min()
    resistance = df["High"].tail(60).max()
    return dict(
        close=c, ema20=e20, sma50=s50, sma200=s200, rsi=r, atr=a,
        score=pts / 7 * 100, pos=notes_pos, neg=notes_neg,
        support=support, resistance=resistance,
    )


def build_plan(tech, fund_pct, info):
    combined = 0.5 * fund_pct + 0.5 * tech["score"]
    c, r, a = tech["close"], tech["rsi"], tech["atr"]
    pullback = tech["ema20"] if tech["ema20"] < c * 0.99 else tech["support"]

    if combined >= 60 and r < 70 and c > tech["sma50"]:
        signal, entry = "🟢 ซื้อ (ทยอยเข้า)", c
    elif combined >= 60:
        signal, entry = "🟡 รอจังหวะย่อแล้วค่อยซื้อ", pullback
    elif combined >= 45:
        signal, entry = "🟡 ถือ / รอดูสัญญาณ", pullback
    else:
        signal, entry = "🔴 หลีกเลี่ยง / ขาย", pullback

    stop = entry - 2 * a
    risk = entry - stop
    t1, t2 = entry + 1.5 * risk, entry + 3 * risk
    analyst = g(info, "targetMeanPrice")
    return dict(
        combined=combined, signal=signal, entry=entry,
        zone=(entry - 0.5 * a, entry + 0.5 * a),
        stop=stop, t1=t1, t2=t2, analyst=analyst,
        rr1=1.5, rr2=3.0,
    )


def analyze(symbol):
    hist, info = load(symbol)
    if hist is None or len(hist) < 210:
        return None
    df = add_indicators(hist.dropna(subset=["Close"]))
    price = df["Close"].iloc[-1]
    rows, fpct = score_fundamentals(info, price)
    tech = analyze_technical(df)
    plan = build_plan(tech, fpct, info)
    return dict(df=df, info=info, rows=rows, fpct=fpct, tech=tech, plan=plan, price=price)


# ----------------------------------------------------------------- chart
def make_chart(df, plan, symbol, days):
    d = df.tail(days)
    fig = make_subplots(rows=2, cols=1, shared_xaxes=True, row_heights=[0.75, 0.25], vertical_spacing=0.03)
    fig.add_trace(go.Candlestick(
        x=d.index, open=d["Open"], high=d["High"], low=d["Low"], close=d["Close"], name="ราคา"), row=1, col=1)
    for col, color in [("EMA20", "#f6c344"), ("SMA50", "#3b82f6"), ("SMA200", "#a855f7")]:
        fig.add_trace(go.Scatter(x=d.index, y=d[col], name=col, line=dict(width=1.3, color=color)), row=1, col=1)
    fig.add_hrect(y0=plan["zone"][0], y1=plan["zone"][1], fillcolor="green", opacity=0.15,
                  line_width=0, row=1, col=1)
    for y, label, color in [(plan["stop"], "Stop", "red"), (plan["t1"], "เป้า 1", "green"),
                            (plan["t2"], "เป้า 2", "darkgreen")]:
        fig.add_hline(y=y, line_dash="dash", line_color=color, annotation_text=f"{label} {y:.2f}",
                      annotation_position="right", row=1, col=1)
    fig.add_trace(go.Scatter(x=d.index, y=d["RSI"], name="RSI", line=dict(color="#ef4444")), row=2, col=1)
    fig.add_hline(y=70, line_dash="dot", row=2, col=1)
    fig.add_hline(y=30, line_dash="dot", row=2, col=1)
    fig.update_layout(height=650, xaxis_rangeslider_visible=False, margin=dict(l=10, r=10, t=30, b=10),
                      title=symbol, legend=dict(orientation="h"))
    return fig
